-- 6. Dashboard summary (Total / Open / In Progress / Resolved)
CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_Dashboard_GetTotalCount]
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        COUNT(*) AS TotalRequests,
        SUM(CASE WHEN Status = 'Open'        THEN 1 ELSE 0 END) AS OpenRequests,
        SUM(CASE WHEN Status = 'In Progress' THEN 1 ELSE 0 END) AS InProgressRequests,
        SUM(CASE WHEN Status = 'Resolved'    THEN 1 ELSE 0 END) AS ResolvedRequests
    FROM Requests
    WHERE IsActive = 1;
END;
GO

-- 7. Update (edit) a request
CREATE PROCEDURE [dbo].[SSR_Request_Update]
(
    @RequestId          INT,
    @RequestType        NVARCHAR(100),
    @RequestDescription NVARCHAR(1000),
    @Priority            NVARCHAR(20),
    @Status               NVARCHAR(30)
)
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE Requests
    SET RequestType        = @RequestType,
        RequestDescription = @RequestDescription,
        Priority            = @Priority,
        Status              = @Status,
        UpdatedDate         = SYSUTCDATETIME()
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    SELECT
        RequestId,
        RequestCode,
        RequestType,
        RequestDescription,
        Priority,
        Status,
        CreatedDate,
        UpdatedDate
    FROM Requests
    WHERE RequestId = @RequestId;
END;
GO

-- 8. Soft delete a request
CREATE PROCEDURE [dbo].[SSR_Request_SoftDelete]
(
    @RequestId INT
)
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE Requests
    SET IsActive = 0,
        UpdatedDate = SYSUTCDATETIME()
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    SELECT @@ROWCOUNT AS RowsAffected;
END;

CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_AssignTechnician]
(
    @RequestId INT,
    @TechnicianId INT
)
AS
BEGIN

    SET NOCOUNT ON;

    -- Validate request
    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.Requests
        WHERE RequestId = @RequestId
          AND IsActive = 1
    )
    BEGIN
        SELECT
            'REQUEST_NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    -- Validate technician
    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.Technicians
        WHERE TechnicianId = @TechnicianId
          AND IsActive = 1
    )
    BEGIN
        SELECT
            'TECHNICIAN_NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    -- Assign technician
    UPDATE dbo.Requests
    SET
        TechnicianId = @TechnicianId,
        UpdatedDate = SYSUTCDATETIME()
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    -- Return updated request
    SELECT
        r.RequestId,
        r.RequestCode,
        r.RequestType,
        r.RequestDescription,
        r.Priority,
        r.Status,
        r.TechnicianId,
        t.TechnicianName,
        r.CreatedDate,
        r.UpdatedDate
    FROM dbo.Requests r
    LEFT JOIN dbo.Technicians t
        ON r.TechnicianId = t.TechnicianId
    WHERE r.RequestId = @RequestId
      AND r.IsActive = 1;

END;
GO

-- Register Employee
--------------------------------------------------

CREATE OR ALTER PROCEDURE [dbo].[SSR_User_RegisterEmployee]
(
    @Username NVARCHAR(50),
    @Email NVARCHAR(150),
    @Password NVARCHAR(100)
)
AS
BEGIN
    SET NOCOUNT ON;

    -- Username already exists
    IF EXISTS
    (
        SELECT 1
        FROM dbo.Users
        WHERE Username = @Username
    )
    BEGIN
        SELECT
            CAST(0 AS bit) AS Success,
            'USERNAME_EXISTS' AS ErrorCode;
        RETURN;
    END;

    -- Email already exists
    IF EXISTS
    (
        SELECT 1
        FROM dbo.Users
        WHERE Email = @Email
    )
    BEGIN
        SELECT
            CAST(0 AS bit) AS Success,
            'EMAIL_EXISTS' AS ErrorCode;
        RETURN;
    END;

    -- Get Employee role
    DECLARE @EmployeeRoleId INT;

    SELECT
        @EmployeeRoleId = RoleId
    FROM dbo.Roles
    WHERE RoleName = 'Employee'
      AND IsActive = 1;

    -- Employee role not available
    IF @EmployeeRoleId IS NULL
    BEGIN
        SELECT
            CAST(0 AS bit) AS Success,
            'EMPLOYEE_ROLE_NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    -- Create employee
    INSERT INTO dbo.Users
    (
        Username,
        Email,
        Password,
        RoleId,
        IsActive,
        CreatedDate
    )
    VALUES
    (
        @Username,
        @Email,
        @Password,
        @EmployeeRoleId,
        1,
        SYSUTCDATETIME()
    );

    -- Registration successful
    SELECT
        CAST(1 AS bit) AS Success,
        CAST(NULL AS NVARCHAR(50)) AS ErrorCode;
END;
GO

--------------------------------------------------
-- 4. Create technician: Users row + Technicians row
--    (same result set as before: Success, ErrorCode)
--------------------------------------------------
CREATE OR ALTER PROCEDURE [dbo].[SSR_User_CreateTechnician]
(
    @Username NVARCHAR(50),
    @Email NVARCHAR(150),
    @Password NVARCHAR(100)
)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    -- Check username
    IF EXISTS (SELECT 1 FROM dbo.Users WHERE Username = @Username)
    BEGIN
        SELECT CAST(0 AS bit) AS Success, 'USERNAME_EXISTS' AS ErrorCode;
        RETURN;
    END;

    -- Check email
    IF EXISTS (SELECT 1 FROM dbo.Users WHERE Email = @Email)
    BEGIN
        SELECT CAST(0 AS bit) AS Success, 'EMAIL_EXISTS' AS ErrorCode;
        RETURN;
    END;

    -- Get Technician role
    DECLARE @TechnicianRoleId INT;

    SELECT @TechnicianRoleId = RoleId
    FROM dbo.Roles
    WHERE RoleName = 'Technician'
      AND IsActive = 1;

    IF @TechnicianRoleId IS NULL
    BEGIN
        SELECT CAST(0 AS bit) AS Success, 'TECHNICIAN_ROLE_NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    BEGIN TRY
        BEGIN TRANSACTION;

        INSERT INTO dbo.Users
        (
            Username, Email, Password, RoleId, IsActive, CreatedDate
        )
        VALUES
        (
            @Username, @Email, @Password, @TechnicianRoleId, 1, SYSUTCDATETIME()
        );

        DECLARE @NewUserId INT = SCOPE_IDENTITY();

        INSERT INTO dbo.Technicians (TechnicianName, IsActive, UserId)
        VALUES (@Username, 1, @NewUserId);

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        THROW;
    END CATCH;

    SELECT CAST(1 AS bit) AS Success, CAST(NULL AS NVARCHAR(50)) AS ErrorCode;
END;
GO

--------------------------------------------------
-- 5. Update status: ownership + transition rules + history
--    Old calls (no role) are treated as Admin
--------------------------------------------------
CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_UpdateStatus]
(
    @RequestId INT,
    @Status NVARCHAR(30),
    @CallerUserId INT = NULL,
    @CallerRole NVARCHAR(50) = NULL,
    @Remarks NVARCHAR(500) = NULL
)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    DECLARE @Role NVARCHAR(50) = ISNULL(@CallerRole, 'Admin');
    DECLARE @OldStatus NVARCHAR(30);

    SELECT @OldStatus = Status
    FROM dbo.Requests
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    -- Request not found or inactive: return nothing
    IF @OldStatus IS NULL
        RETURN;

    -- Technician: only for requests assigned to them
    IF @Role = 'Technician'
       AND NOT EXISTS
       (
           SELECT 1
           FROM dbo.Requests r
           INNER JOIN dbo.Technicians t ON r.TechnicianId = t.TechnicianId
           WHERE r.RequestId = @RequestId
             AND r.IsActive = 1
             AND t.IsActive = 1
             AND t.UserId = @CallerUserId
       )
    BEGIN
        SELECT 'NOT_ASSIGNED' AS ErrorCode, @OldStatus AS CurrentStatus;
        RETURN;
    END;

    -- Employee: only for their own requests
    IF @Role = 'Employee'
       AND NOT EXISTS
       (
           SELECT 1
           FROM dbo.Requests
           WHERE RequestId = @RequestId
             AND IsActive = 1
             AND CreatedByUserId = @CallerUserId
       )
    BEGIN
        SELECT 'NOT_OWNER' AS ErrorCode, @OldStatus AS CurrentStatus;
        RETURN;
    END;

    -- The move must be allowed for this role
    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.RequestStatusTransitions
        WHERE RoleName = @Role
          AND FromStatus = @OldStatus
          AND ToStatus = @Status
    )
    BEGIN
        SELECT 'INVALID_TRANSITION' AS ErrorCode, @OldStatus AS CurrentStatus;
        RETURN;
    END;

    BEGIN TRY
        BEGIN TRANSACTION;

        UPDATE dbo.Requests
        SET Status = @Status,
            UpdatedDate = SYSUTCDATETIME()
        WHERE RequestId = @RequestId
          AND IsActive = 1
          AND Status = @OldStatus;

        -- Somebody changed the request at the same moment
        IF @@ROWCOUNT = 0
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT 'INVALID_TRANSITION' AS ErrorCode, @OldStatus AS CurrentStatus;
            RETURN;
        END;

        INSERT INTO dbo.RequestStatusHistory
            (RequestId, OldStatus, NewStatus, ChangedByUserId, ChangedByRole, Remarks)
        VALUES
            (@RequestId, @OldStatus, @Status, @CallerUserId, @Role, @Remarks);

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        THROW;
    END CATCH;

    SELECT
        r.RequestId,
        r.RequestCode,
        r.RequestType,
        r.RequestDescription,
        r.Priority,
        r.Status,
        r.TechnicianId,
        t.TechnicianName,
        r.CreatedDate,
        r.UpdatedDate
    FROM dbo.Requests r
    LEFT JOIN dbo.Technicians t
        ON r.TechnicianId = t.TechnicianId
    WHERE r.RequestId = @RequestId
      AND r.IsActive = 1;
END;
GO

CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_Update]
(
    @RequestId          INT,
    @RequestType        NVARCHAR(100),
    @RequestDescription NVARCHAR(1000),
    @Priority           NVARCHAR(20),
    @Status             NVARCHAR(30),
    @CallerUserId       INT = NULL,
    @CallerRole         NVARCHAR(50) = NULL
)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    DECLARE @Role NVARCHAR(50) = ISNULL(@CallerRole, 'Admin');
    DECLARE @OldStatus NVARCHAR(30);

    SELECT @OldStatus = Status
    FROM dbo.Requests
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    -- Request not found or inactive: return nothing
    IF @OldStatus IS NULL
        RETURN;

    -- Status changed: the move must be allowed
    IF @OldStatus <> @Status
       AND NOT EXISTS
       (
           SELECT 1
           FROM dbo.RequestStatusTransitions
           WHERE RoleName = @Role
             AND FromStatus = @OldStatus
             AND ToStatus = @Status
       )
    BEGIN
        SELECT 'INVALID_TRANSITION' AS ErrorCode, @OldStatus AS CurrentStatus;
        RETURN;
    END;

    BEGIN TRY
        BEGIN TRANSACTION;

        UPDATE dbo.Requests
        SET RequestType        = @RequestType,
            RequestDescription = @RequestDescription,
            Priority           = @Priority,
            Status             = @Status,
            UpdatedDate        = SYSUTCDATETIME()
        WHERE RequestId = @RequestId
          AND IsActive = 1
          AND Status = @OldStatus;

        -- Somebody changed the request at the same moment
        IF @@ROWCOUNT = 0
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT 'INVALID_TRANSITION' AS ErrorCode, @OldStatus AS CurrentStatus;
            RETURN;
        END;

        IF @OldStatus <> @Status
        BEGIN
            INSERT INTO dbo.RequestStatusHistory
                (RequestId, OldStatus, NewStatus, ChangedByUserId, ChangedByRole)
            VALUES
                (@RequestId, @OldStatus, @Status, @CallerUserId, @Role);
        END;

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        THROW;
    END CATCH;

    SELECT
        RequestId,
        RequestCode,
        RequestType,
        RequestDescription,
        Priority,
        Status,
        CreatedDate,
        UpdatedDate
    FROM Requests
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    @RequestId INT,

    @CallerRole NVARCHAR(50) = NULL
)

BEGIN
    SET NOCOUNT ON;

    DECLARE @CurrentStatus NVARCHAR(30);

    SELECT @CurrentStatus = Status
    FROM dbo.Requests
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    IF @CurrentStatus IS NULL
        RETURN;

    -- Technician sees options only for requests assigned to them
    IF @Role = 'Technician'
       AND NOT EXISTS
       (
           SELECT 1
           FROM dbo.Requests r
           INNER JOIN dbo.Technicians t ON r.TechnicianId = t.TechnicianId
           WHERE r.RequestId = @RequestId
             AND r.IsActive = 1
             AND t.IsActive = 1
             AND t.UserId = @CallerUserId
       )
        RETURN;

       AND NOT EXISTS
       (
           SELECT 1
           FROM dbo.Requests
           WHERE RequestId = @RequestId
             AND IsActive = 1
             AND CreatedByUserId = @CallerUserId
       )
        RETURN;

    FROM dbo.RequestStatusTransitions

      AND FromStatus = @CurrentStatus

                 WHEN 'In Progress' THEN 1
                 WHEN 'On Hold'     THEN 2

USE [SmartServiceRequestDB];
GO

CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_GetPaged]
(
    @Search          NVARCHAR(100) = NULL,
    @Status          NVARCHAR(30)  = NULL,
    @Priority        NVARCHAR(20)  = NULL,
    @CreatedByUserId INT           = NULL,   -- NULL = all requests (Admin), value = "my requests"
    @SortBy          NVARCHAR(30)  = 'CreatedDate',
    @SortDir         NVARCHAR(4)   = 'DESC',
    @PageNumber      INT           = 1,
    @PageSize        INT           = 10
)
AS
BEGIN
    SET NOCOUNT ON;

    -- Safe values
    IF @PageNumber < 1 SET @PageNumber = 1;
    IF @PageSize < 1 SET @PageSize = 10;
    IF @PageSize > 100 SET @PageSize = 100;
    SET @SortDir = CASE WHEN UPPER(ISNULL(@SortDir, '')) = 'ASC' THEN 'ASC' ELSE 'DESC' END;

    SET @Status   = NULLIF(LTRIM(RTRIM(ISNULL(@Status, ''))), '');
    SET @Priority = NULLIF(LTRIM(RTRIM(ISNULL(@Priority, ''))), '');
    SET @Search   = NULLIF(LTRIM(RTRIM(ISNULL(@Search, ''))), '');

    -- Search text: escape LIKE wildcards so "%" or "_" are searched as normal characters
    DECLARE @Like NVARCHAR(210) = NULL;

    IF @Search IS NOT NULL
        SET @Like = '%' +
            REPLACE(REPLACE(REPLACE(REPLACE(@Search, '\', '\\'), '%', '\%'), '_', '\_'), '[', '\[')
            + '%';

    -- 1. Total rows for the current filters
    DECLARE @Total INT;

    SELECT @Total = COUNT(*)
    FROM dbo.Requests r
    WHERE r.IsActive = 1
      AND (@CreatedByUserId IS NULL OR r.CreatedByUserId = @CreatedByUserId)
      AND (@Status   IS NULL OR r.Status   = @Status)
      AND (@Priority IS NULL OR r.Priority = @Priority)
      AND (   @Like IS NULL
           OR r.RequestCode        LIKE @Like ESCAPE '\'
           OR r.RequestType        LIKE @Like ESCAPE '\'
           OR r.RequestDescription LIKE @Like ESCAPE '\');

    -- 2. If the page is beyond the last page (e.g. last row deleted), go to the last page
    DECLARE @TotalPages INT =
        CASE WHEN @Total = 0 THEN 0 ELSE (@Total + @PageSize - 1) / @PageSize END;

    IF @TotalPages > 0 AND @PageNumber > @TotalPages
        SET @PageNumber = @TotalPages;

    -- Result set 1: paging info
    SELECT
        @Total      AS TotalCount,
        @PageNumber AS PageNumber,
        @PageSize   AS PageSize,
        @TotalPages AS TotalPages;

    -- Result set 2: rows (same columns as SSR_Request_GetAll)
    SELECT
        r.RequestId,
        r.RequestCode,
        r.RequestType,
        r.RequestDescription,
        r.Priority,
        r.Status,
        r.CreatedDate,
        r.UpdatedDate
    FROM dbo.Requests r
    WHERE r.IsActive = 1
      AND (@CreatedByUserId IS NULL OR r.CreatedByUserId = @CreatedByUserId)
      AND (@Status   IS NULL OR r.Status   = @Status)
      AND (@Priority IS NULL OR r.Priority = @Priority)
      AND (   @Like IS NULL
           OR r.RequestCode        LIKE @Like ESCAPE '\'
           OR r.RequestType        LIKE @Like ESCAPE '\'
           OR r.RequestDescription LIKE @Like ESCAPE '\')
    ORDER BY
        -- Only whitelisted columns are sorted (no dynamic SQL)
        CASE WHEN @SortBy = 'RequestCode' AND @SortDir = 'ASC'  THEN r.RequestId END ASC,
        CASE WHEN @SortBy = 'RequestCode' AND @SortDir = 'DESC' THEN r.RequestId END DESC,

        CASE WHEN @SortBy = 'RequestType' AND @SortDir = 'ASC'  THEN r.RequestType END ASC,
        CASE WHEN @SortBy = 'RequestType' AND @SortDir = 'DESC' THEN r.RequestType END DESC,

        -- Priority sorts by importance: Low < Medium < High < Critical
        CASE WHEN @SortBy = 'Priority' AND @SortDir = 'ASC' THEN
            CASE r.Priority WHEN 'Low' THEN 1 WHEN 'Medium' THEN 2 WHEN 'High' THEN 3 WHEN 'Critical' THEN 4 ELSE 0 END
        END ASC,
        CASE WHEN @SortBy = 'Priority' AND @SortDir = 'DESC' THEN
            CASE r.Priority WHEN 'Low' THEN 1 WHEN 'Medium' THEN 2 WHEN 'High' THEN 3 WHEN 'Critical' THEN 4 ELSE 0 END
        END DESC,

        CASE WHEN @SortBy = 'Status' AND @SortDir = 'ASC'  THEN r.Status END ASC,
        CASE WHEN @SortBy = 'Status' AND @SortDir = 'DESC' THEN r.Status END DESC,

        CASE WHEN @SortBy = 'CreatedDate' AND @SortDir = 'ASC'  THEN r.CreatedDate END ASC,
        CASE WHEN @SortBy = 'CreatedDate' AND @SortDir = 'DESC' THEN r.CreatedDate END DESC,

        -- Default order and tie-breaker (stable paging)
        r.CreatedDate DESC,
        r.RequestId DESC
    OFFSET (@PageNumber - 1) * @PageSize ROWS
    FETCH NEXT @PageSize ROWS ONLY;
END;
GO

        CASE WHEN @SortBy = 'RoleName' AND @SortDir = 'DESC' THEN r.RoleName END DESC,

        CASE WHEN @SortBy = 'IsActive' AND @SortDir = 'ASC'  THEN u.IsActive END ASC,
        CASE WHEN @SortBy = 'IsActive' AND @SortDir = 'DESC' THEN u.IsActive END DESC,

        CASE WHEN @SortBy = 'CreatedDate' AND @SortDir = 'ASC'  THEN u.CreatedDate END ASC,
        CASE WHEN @SortBy = 'CreatedDate' AND @SortDir = 'DESC' THEN u.CreatedDate END DESC,

        -- Default order and tie-breaker (stable paging)
        u.CreatedDate DESC,
        u.UserId DESC
    OFFSET (@PageNumber - 1) * @PageSize ROWS
    FETCH NEXT @PageSize ROWS ONLY;
END;

-- 2. Activate / deactivate a user
--    Rules: not yourself, not an Admin account.
--    A technician's Technicians row follows the user,
--    so an inactive technician cannot be assigned.
--------------------------------------------------
CREATE OR ALTER PROCEDURE [dbo].[SSR_User_SetActive]
(
    @UserId       INT,
    @IsActive     BIT,
    @CallerUserId INT
)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    DECLARE @RoleName NVARCHAR(50);

    SELECT @RoleName = r.RoleName
    FROM dbo.Users u
    INNER JOIN dbo.Roles r ON r.RoleId = u.RoleId
    WHERE u.UserId = @UserId;

    IF @RoleName IS NULL
    BEGIN
        SELECT 'USER_NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    IF @UserId = @CallerUserId
    BEGIN
        SELECT 'CANNOT_CHANGE_SELF' AS ErrorCode;
        RETURN;
    END;

    IF @RoleName = 'Admin'
    BEGIN
        SELECT 'ADMIN_NOT_ALLOWED' AS ErrorCode;
        RETURN;
    END;

    BEGIN TRY
        BEGIN TRANSACTION;

        UPDATE dbo.Users
        SET IsActive    = @IsActive,
            UpdatedDate = SYSUTCDATETIME()
        WHERE UserId = @UserId;

        IF @RoleName = 'Technician'
        BEGIN
            UPDATE dbo.Technicians
            SET IsActive = @IsActive
            WHERE UserId = @UserId;
        END;

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        THROW;
    END CATCH;

    SELECT
        u.UserId,
        u.Username,
        u.Email,
        r.RoleName,
        u.IsActive,
        u.CreatedDate
    FROM dbo.Users u
    INNER JOIN dbo.Roles r ON r.RoleId = u.RoleId
    WHERE u.UserId = @UserId;

-- 1. Assigned requests of the logged-in technician
--    (same search / filter / sort / paging pattern as step 2)

CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_GetAssignedPaged]
(
    @TechnicianUserId INT,
    @Search           NVARCHAR(100) = NULL,
    @Status           NVARCHAR(30)  = NULL,
    @Priority         NVARCHAR(20)  = NULL,
    @SortBy           NVARCHAR(30)  = 'CreatedDate',
    @SortDir          NVARCHAR(4)   = 'DESC',
    @PageNumber       INT           = 1,
    @PageSize         INT           = 10
)
AS
BEGIN
    SET NOCOUNT ON;

    IF @PageNumber < 1 SET @PageNumber = 1;
    IF @PageSize < 1 SET @PageSize = 10;
    IF @PageSize > 100 SET @PageSize = 100;
    SET @SortDir = CASE WHEN UPPER(ISNULL(@SortDir, '')) = 'ASC' THEN 'ASC' ELSE 'DESC' END;

    SET @Status   = NULLIF(LTRIM(RTRIM(ISNULL(@Status, ''))), '');
    SET @Priority = NULLIF(LTRIM(RTRIM(ISNULL(@Priority, ''))), '');
    SET @Search   = NULLIF(LTRIM(RTRIM(ISNULL(@Search, ''))), '');

    DECLARE @Like NVARCHAR(210) = NULL;

    IF @Search IS NOT NULL
        SET @Like = '%' +
            REPLACE(REPLACE(REPLACE(REPLACE(@Search, '\', '\\'), '%', '\%'), '_', '\_'), '[', '\[')
            + '%';

    DECLARE @Total INT;

    SELECT @Total = COUNT(*)
    FROM dbo.Requests r
    INNER JOIN dbo.Technicians t
        ON t.TechnicianId = r.TechnicianId
       AND t.UserId = @TechnicianUserId
       AND t.IsActive = 1
    WHERE r.IsActive = 1
      AND (@Status   IS NULL OR r.Status   = @Status)
      AND (@Priority IS NULL OR r.Priority = @Priority)
      AND (   @Like IS NULL
           OR r.RequestCode        LIKE @Like ESCAPE '\'
           OR r.RequestType        LIKE @Like ESCAPE '\'
           OR r.RequestDescription LIKE @Like ESCAPE '\');

    DECLARE @TotalPages INT =
        CASE WHEN @Total = 0 THEN 0 ELSE (@Total + @PageSize - 1) / @PageSize END;

    IF @TotalPages > 0 AND @PageNumber > @TotalPages
        SET @PageNumber = @TotalPages;

    -- Result set 1: paging info
    SELECT
        @Total      AS TotalCount,
        @PageNumber AS PageNumber,
        @PageSize   AS PageSize,
        @TotalPages AS TotalPages;

    -- Result set 2: rows (same columns as SSR_Request_GetPaged)
    SELECT
        r.RequestId,
        r.RequestCode,
        r.RequestType,
        r.RequestDescription,
        r.Priority,
        r.Status,
        r.CreatedDate,
        r.UpdatedDate
    FROM dbo.Requests r
    INNER JOIN dbo.Technicians t
        ON t.TechnicianId = r.TechnicianId
       AND t.UserId = @TechnicianUserId
       AND t.IsActive = 1
    WHERE r.IsActive = 1
      AND (@Status   IS NULL OR r.Status   = @Status)
      AND (@Priority IS NULL OR r.Priority = @Priority)
      AND (   @Like IS NULL
           OR r.RequestCode        LIKE @Like ESCAPE '\'
           OR r.RequestType        LIKE @Like ESCAPE '\'
           OR r.RequestDescription LIKE @Like ESCAPE '\')
    ORDER BY
        CASE WHEN @SortBy = 'RequestCode' AND @SortDir = 'ASC'  THEN r.RequestId END ASC,
        CASE WHEN @SortBy = 'RequestCode' AND @SortDir = 'DESC' THEN r.RequestId END DESC,

        CASE WHEN @SortBy = 'RequestType' AND @SortDir = 'ASC'  THEN r.RequestType END ASC,
        CASE WHEN @SortBy = 'RequestType' AND @SortDir = 'DESC' THEN r.RequestType END DESC,

        CASE WHEN @SortBy = 'Priority' AND @SortDir = 'ASC' THEN
            CASE r.Priority WHEN 'Low' THEN 1 WHEN 'Medium' THEN 2 WHEN 'High' THEN 3 WHEN 'Critical' THEN 4 ELSE 0 END
        END ASC,
        CASE WHEN @SortBy = 'Priority' AND @SortDir = 'DESC' THEN
            CASE r.Priority WHEN 'Low' THEN 1 WHEN 'Medium' THEN 2 WHEN 'High' THEN 3 WHEN 'Critical' THEN 4 ELSE 0 END
        END DESC,

        CASE WHEN @SortBy = 'Status' AND @SortDir = 'ASC'  THEN r.Status END ASC,
        CASE WHEN @SortBy = 'Status' AND @SortDir = 'DESC' THEN r.Status END DESC,

        CASE WHEN @SortBy = 'CreatedDate' AND @SortDir = 'ASC'  THEN r.CreatedDate END ASC,
        CASE WHEN @SortBy = 'CreatedDate' AND @SortDir = 'DESC' THEN r.CreatedDate END DESC,

        r.CreatedDate DESC,
        r.RequestId DESC
    OFFSET (@PageNumber - 1) * @PageSize ROWS
    FETCH NEXT @PageSize ROWS ONLY;
END;

CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_GetStatusHistory]
(
    @RequestId    INT,
    @CallerUserId INT,
    @CallerRole   NVARCHAR(50)
)
AS
BEGIN
    SET NOCOUNT ON;

    -- Same visibility as the request itself:
    --   Admin      : any active request
    --   Employee   : only requests they created
    --   Technician : only requests assigned to them
    -- Anything else returns one NOT_FOUND row (existence is not revealed).
    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.Requests r
        WHERE r.RequestId = @RequestId
          AND r.IsActive = 1
          AND
          (
              @CallerRole = 'Admin'
              OR (@CallerRole = 'Employee' AND r.CreatedByUserId = @CallerUserId)
              OR (@CallerRole = 'Technician' AND EXISTS
                  (
                      SELECT 1
                      FROM dbo.Technicians t
                      WHERE t.TechnicianId = r.TechnicianId
                        AND t.IsActive = 1
                        AND t.UserId = @CallerUserId
                  ))
          )
    )
    BEGIN
        SELECT 'NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    -- Oldest first. Rows written by older procedures have no role saved,
    -- so the role is taken from the user when it is missing.
    SELECT
        h.HistoryId,
        h.RequestId,
        h.OldStatus,
        h.NewStatus,
        h.ChangedByUserId,
        u.Username                          AS ChangedByUsername,
        ISNULL(h.ChangedByRole, ro.RoleName) AS ChangedByRole,
        h.Remarks,
        h.ChangedDate
    FROM dbo.RequestStatusHistory h
    LEFT JOIN dbo.Users u  ON u.UserId  = h.ChangedByUserId
    LEFT JOIN dbo.Roles ro ON ro.RoleId = u.RoleId
    WHERE h.RequestId = @RequestId
    ORDER BY h.ChangedDate, h.HistoryId;
END;

-- 3. Count a wrong password; lock the account when the limit is reached
CREATE OR ALTER PROCEDURE [dbo].[SSR_User_RecordFailedLogin]
(
    @UserId         INT,
    @MaxAttempts    INT,
    @LockoutMinutes INT
)
AS
BEGIN
    SET NOCOUNT ON;

    -- One atomic UPDATE: on the Nth failure the counter restarts at 0
    -- and LockoutEndUtc is set, so the user gets a fresh set of attempts
    -- after the lock expires.
    UPDATE dbo.Users
    SET FailedLoginCount =
            CASE WHEN FailedLoginCount + 1 >= @MaxAttempts
                 THEN 0
                 ELSE FailedLoginCount + 1 END,
        LockoutEndUtc =
            CASE WHEN FailedLoginCount + 1 >= @MaxAttempts
                 THEN DATEADD(MINUTE, @LockoutMinutes, SYSUTCDATETIME())
                 ELSE LockoutEndUtc END
    WHERE UserId = @UserId;

    SELECT LockoutEndUtc
    FROM dbo.Users
    WHERE UserId = @UserId;
END;
GO

-- 4. Clear the counter after a successful login
CREATE OR ALTER PROCEDURE [dbo].[SSR_User_ResetFailedLogin]
(
    @UserId INT
)
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE dbo.Users
    SET FailedLoginCount = 0,
        LockoutEndUtc = NULL
    WHERE UserId = @UserId
      AND (FailedLoginCount <> 0 OR LockoutEndUtc IS NOT NULL);
END;

-- 5. Save a technician's profile and specializations in one transaction
--    @SpecializationIds is a comma-separated list, for example '1,4'
CREATE OR ALTER PROCEDURE [dbo].[SSR_TechnicianProfile_Save]
(
    @UserId            INT,
    @FullName          NVARCHAR(100),
    @Phone             NVARCHAR(20) = NULL,
    @SpecializationIds NVARCHAR(500)
)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    DECLARE @TechnicianId INT;

    SELECT @TechnicianId = TechnicianId
    FROM dbo.Technicians
    WHERE UserId = @UserId
      AND IsActive = 1;

    IF @TechnicianId IS NULL
    BEGIN
        SELECT CAST(0 AS BIT) AS Success, 'TECHNICIAN_NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    DECLARE @Wanted TABLE (SpecializationId INT PRIMARY KEY);

    INSERT INTO @Wanted (SpecializationId)
    SELECT DISTINCT TRY_CAST(LTRIM(RTRIM([value])) AS INT)
    FROM STRING_SPLIT(@SpecializationIds, ',')
    WHERE TRY_CAST(LTRIM(RTRIM([value])) AS INT) IS NOT NULL;

    IF NOT EXISTS (SELECT 1 FROM @Wanted)
    BEGIN
        SELECT CAST(0 AS BIT) AS Success, 'NO_SPECIALIZATION' AS ErrorCode;
        RETURN;
    END;

    -- Every chosen id must be an active specialization
    IF EXISTS
    (
        SELECT 1
        FROM @Wanted w
        WHERE NOT EXISTS
        (
            SELECT 1
            FROM dbo.Specializations s
            WHERE s.SpecializationId = w.SpecializationId
              AND s.IsActive = 1
        )
    )
    BEGIN
        SELECT CAST(0 AS BIT) AS Success, 'INVALID_SPECIALIZATION' AS ErrorCode;
        RETURN;
    END;

    BEGIN TRANSACTION;

    UPDATE dbo.UserProfiles WITH (UPDLOCK, SERIALIZABLE)
    SET FullName    = @FullName,
        Phone       = NULLIF(LTRIM(RTRIM(@Phone)), N''),
        UpdatedDate = SYSUTCDATETIME()
    WHERE UserId = @UserId;

    IF @@ROWCOUNT = 0
    BEGIN
        INSERT INTO dbo.UserProfiles (UserId, FullName, Phone)
        VALUES (@UserId, @FullName, NULLIF(LTRIM(RTRIM(@Phone)), N''));
    END;

    DELETE ts
    FROM dbo.TechnicianSpecializations ts
    WHERE ts.TechnicianId = @TechnicianId
      AND ts.SpecializationId NOT IN (SELECT SpecializationId FROM @Wanted);

    INSERT INTO dbo.TechnicianSpecializations (TechnicianId, SpecializationId)
    SELECT @TechnicianId, w.SpecializationId
    FROM @Wanted w
    WHERE NOT EXISTS
    (
        SELECT 1
        FROM dbo.TechnicianSpecializations ts
        WHERE ts.TechnicianId = @TechnicianId
          AND ts.SpecializationId = w.SpecializationId
    );

    COMMIT TRANSACTION;

    SELECT CAST(1 AS BIT) AS Success, CAST(NULL AS VARCHAR(30)) AS ErrorCode;
END;
GO

-- Assign a technician. @MaxActiveRequests = NULL means "no workload check" (the old behaviour).
CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_AssignTechnician]
(
    @RequestId         INT,
    @TechnicianId      INT,
    @MaxActiveRequests INT = NULL
)
AS
BEGIN

    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    -- Validate request
    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.Requests
        WHERE RequestId = @RequestId
          AND IsActive = 1
    )
    BEGIN
        SELECT
            'REQUEST_NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    -- Validate technician
    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.Technicians
        WHERE TechnicianId = @TechnicianId
          AND IsActive = 1
    )
    BEGIN
        SELECT
            'TECHNICIAN_NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    BEGIN TRANSACTION;

    -- Lock this technician's row, so two admins cannot both take the last free slot
    DECLARE @LockedTechnicianId INT;

    SELECT @LockedTechnicianId = TechnicianId
    FROM dbo.Technicians WITH (UPDLOCK, ROWLOCK)
    WHERE TechnicianId = @TechnicianId;

    IF @MaxActiveRequests IS NOT NULL
    BEGIN
        DECLARE @CurrentTechnicianId INT;
        DECLARE @CurrentStatus NVARCHAR(30);

        SELECT
            @CurrentTechnicianId = TechnicianId,
            @CurrentStatus = Status
        FROM dbo.Requests
        WHERE RequestId = @RequestId
          AND IsActive = 1;

        -- Only a request that is still active work adds to the workload,
        -- and moving it to the technician who already has it adds nothing.
        IF @CurrentStatus IN (N'Open', N'In Progress', N'On Hold')
           AND ISNULL(@CurrentTechnicianId, 0) <> @TechnicianId
           AND
           (
               SELECT COUNT(*)
               FROM dbo.Requests
               WHERE TechnicianId = @TechnicianId
                 AND IsActive = 1
                 AND Status IN (N'Open', N'In Progress', N'On Hold')
           ) >= @MaxActiveRequests
        BEGIN
            ROLLBACK TRANSACTION;

            SELECT
                'TECHNICIAN_AT_CAPACITY' AS ErrorCode;
            RETURN;
        END;
    END;

    -- Assign technician
    UPDATE dbo.Requests
    SET
        TechnicianId = @TechnicianId,
        UpdatedDate = SYSUTCDATETIME()
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    COMMIT TRANSACTION;

    -- Return updated request
    SELECT
        r.RequestId,
        r.RequestCode,
        r.RequestType,
        r.RequestDescription,
        r.Priority,
        r.Status,
        r.TechnicianId,
        t.TechnicianName,
        r.CreatedDate,
        r.UpdatedDate
    FROM dbo.Requests r
    LEFT JOIN dbo.Technicians t
        ON r.TechnicianId = t.TechnicianId
    WHERE r.RequestId = @RequestId
      AND r.IsActive = 1;

END;
