-- Stored Procedures
CREATE PROCEDURE [dbo].[SSR_Request_Create]
(
    @RequestType NVARCHAR(100),
    @RequestDescription NVARCHAR(1000),
    @Priority NVARCHAR(20)
)
AS
BEGIN
    SET NOCOUNT ON;

    INSERT INTO Requests
    (
        RequestType,
        RequestDescription,
        Priority
    )
    VALUES
    (
        @RequestType,
        @RequestDescription,
        @Priority
    );

    SELECT
        RequestId,
        RequestCode,
        RequestType,
        RequestDescription,
        Priority,
        Status,
        CreatedDate
    FROM Requests
    WHERE RequestId = SCOPE_IDENTITY();
END;

CREATE PROCEDURE [dbo].[SSR_Request_Dashboard_GetTotalCount]
AS
BEGIN
    SET NOCOUNT ON;

    SELECT COUNT(*) AS TotalRequests
    FROM Requests;
END;

-- 6. Dashboard summary (Total / Open / In Progress / Resolved)
CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_Dashboard_GetTotalCount]
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        COUNT(*) AS TotalRequests,
        SUM(CASE WHEN Status = 'Open'        THEN 1 ELSE 0 END) AS OpenRequests,
        SUM(CASE WHEN Status = 'In Progress' THEN 1 ELSE 0 END) AS InProgressRequests,
        SUM(CASE WHEN Status = 'Resolved'    THEN 1 ELSE 0 END) AS ResolvedRequests
    FROM Requests
    WHERE IsActive = 1;
END;
GO

-- 7. Update (edit) a request
CREATE PROCEDURE [dbo].[SSR_Request_Update]
(
    @RequestId          INT,
    @RequestType        NVARCHAR(100),
    @RequestDescription NVARCHAR(1000),
    @Priority            NVARCHAR(20),
    @Status               NVARCHAR(30)
)
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE Requests
    SET RequestType        = @RequestType,
        RequestDescription = @RequestDescription,
        Priority            = @Priority,
        Status              = @Status,
        UpdatedDate         = SYSUTCDATETIME()
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    SELECT
        RequestId,
        RequestCode,
        RequestType,
        RequestDescription,
        Priority,
        Status,
        CreatedDate,
        UpdatedDate
    FROM Requests
    WHERE RequestId = @RequestId;
END;
GO

-- 8. Soft delete a request
CREATE PROCEDURE [dbo].[SSR_Request_SoftDelete]
(
    @RequestId INT
)
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE Requests
    SET IsActive = 0,
        UpdatedDate = SYSUTCDATETIME()
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    SELECT @@ROWCOUNT AS RowsAffected;
END;

CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_UpdateStatus]
(
    @RequestId INT,
    @Status NVARCHAR(30)
)
AS
BEGIN

    SET NOCOUNT ON;

    UPDATE dbo.Requests
    SET
        Status = @Status,
        UpdatedDate = SYSUTCDATETIME()
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    SELECT
        r.RequestId,
        r.RequestCode,
        r.RequestType,
        r.RequestDescription,
        r.Priority,
        r.Status,
        r.TechnicianId,
        t.TechnicianName,
        r.CreatedDate,
        r.UpdatedDate
    FROM dbo.Requests r
    LEFT JOIN dbo.Technicians t
        ON r.TechnicianId = t.TechnicianId
    WHERE r.RequestId = @RequestId
      AND r.IsActive = 1;

END;
GO


--------------------------------------------------
-- 7. Assign technician
--------------------------------------------------

CREATE OR ALTER PROCEDURE [dbo].[SSR_Request_AssignTechnician]
(
    @RequestId INT,
    @TechnicianId INT
)
AS
BEGIN

    SET NOCOUNT ON;

    -- Validate request
    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.Requests
        WHERE RequestId = @RequestId
          AND IsActive = 1
    )
    BEGIN
        SELECT
            'REQUEST_NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    -- Validate technician
    IF NOT EXISTS
    (
        SELECT 1
        FROM dbo.Technicians
        WHERE TechnicianId = @TechnicianId
          AND IsActive = 1
    )
    BEGIN
        SELECT
            'TECHNICIAN_NOT_FOUND' AS ErrorCode;
        RETURN;
    END;

    -- Assign technician
    UPDATE dbo.Requests
    SET
        TechnicianId = @TechnicianId,
        UpdatedDate = SYSUTCDATETIME()
    WHERE RequestId = @RequestId
      AND IsActive = 1;

    -- Return updated request
    SELECT
        r.RequestId,
        r.RequestCode,
        r.RequestType,
        r.RequestDescription,
        r.Priority,
        r.Status,
        r.TechnicianId,
        t.TechnicianName,
        r.CreatedDate,
        r.UpdatedDate
    FROM dbo.Requests r
    LEFT JOIN dbo.Technicians t
        ON r.TechnicianId = t.TechnicianId
    WHERE r.RequestId = @RequestId
      AND r.IsActive = 1;

END;
GO
