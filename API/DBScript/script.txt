-- Stored Procedures
CREATE PROCEDURE [dbo].[SSR_Request_Create]
(
    @RequestType NVARCHAR(100),
    @RequestDescription NVARCHAR(1000),
    @Priority NVARCHAR(20)
)
AS
BEGIN
    SET NOCOUNT ON;

    INSERT INTO Requests
    (
        RequestType,
        RequestDescription,
        Priority
    )
    VALUES
    (
        @RequestType,
        @RequestDescription,
        @Priority
    );

    SELECT
        RequestId,
        RequestCode,
        RequestType,
        RequestDescription,
        Priority,
        Status,
        CreatedDate
    FROM Requests
    WHERE RequestId = SCOPE_IDENTITY();
END;

CREATE PROCEDURE [dbo].[SSR_Request_Dashboard_GetTotalCount]
AS
BEGIN
    SET NOCOUNT ON;

    SELECT COUNT(*) AS TotalRequests
    FROM Requests;
END;


CREATE PROCEDURE [dbo].[SSR_Request_GetAll]
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        RequestId,
        RequestCode,
        RequestType,
        RequestDescription,
        Priority,
        Status,
        CreatedDate,
        UpdatedDate
    FROM Requests
    ORDER BY CreatedDate DESC;
END;

CREATE PROCEDURE [dbo].[SSR_Request_GetById]
(
    @RequestId INT
)
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        RequestId,
        RequestCode,
        RequestType,
        RequestDescription,
        Priority,
        Status,
        CreatedDate,
        UpdatedDate
    FROM Requests
    WHERE RequestId = @RequestId;
END;
